## 🔍 核心知识点详解

### 索引数组

**定义数组**：
```bash
#!/bin/bash
# 方法 1：直接定义
fruits=(apple banana cherry)

# 方法 2：逐个赋值
fruits[0]="apple"
fruits[1]="banana"
fruits[2]="cherry"

# 方法 3：命令输出
files=($(ls *.txt))

# 方法 4：追加
fruits+=(date)
```

**访问数组**：
```bash
#!/bin/bash
fruits=(apple banana cherry)

# 单个元素
echo ${fruits[0]}      # apple
echo ${fruits[1]}      # banana

# 所有元素
echo ${fruits[@]}      # apple banana cherry
echo ${fruits[*]}      # apple banana cherry

# 数组长度
echo ${#fruits[@]}     # 3

# 元素索引
echo ${!fruits[@]}     # 0 1 2

# 切片
echo ${fruits[@]:1:2}  # banana cherry（从 1 开始取 2 个）
```

**遍历数组**：
```bash
#!/bin/bash
fruits=(apple banana cherry)

# 方法 1：直接遍历
for fruit in "${fruits[@]}"; do
    echo "水果：$fruit"
done

# 方法 2：索引遍历
for i in "${!fruits[@]}"; do
    echo "$i: ${fruits[$i]}"
done

# 方法 3：C 风格
for ((i=0; i<${#fruits[@]}; i++)); do
    echo "$i: ${fruits[$i]}"
done
```

**数组操作**：
```bash
#!/bin/bash
arr1=(1 2 3)
arr2=(4 5 6)

# 合并数组
merged=("${arr1[@]}" "${arr2[@]}")
echo ${merged[@]}  # 1 2 3 4 5 6

# 去重
unique=($(echo "${merged[@]}" | tr ' ' '\n' | sort -u))
echo ${unique[@]}  # 1 2 3 4 5 6

# 搜索元素
search="3"
for i in "${!arr1[@]}"; do
    if [ "${arr1[$i]}" = "$search" ]; then
        echo "找到 $search，索引：$i"
    fi
done
```

### 关联数组

**定义字典**：
```bash
#!/bin/bash
# 声明关联数组（Bash 4.0+）
declare -A user

# 赋值
user[name]="John"
user[age]="25"
user[city]="Beijing"

# 或一次性定义
declare -A user=(
    [name]="John"
    [age]="25"
    [city]="Beijing"
)
```

**访问字典**：
```bash
#!/bin/bash
declare -A user=(
    [name]="John"
    [age]="25"
    [city]="Beijing"
)

# 单个值
echo ${user[name]}     # John
echo ${user[age]}      # 25

# 所有键
echo ${!user[@]}       # name age city

# 所有值
echo ${user[@]}        # John 25 Beijing

# 键数量
echo ${#user[@]}       # 3

# 检查键是否存在
if [[ -v user[name] ]]; then
    echo "name 键存在"
fi
```

**遍历字典**：
```bash
#!/bin/bash
declare -A user=(
    [name]="John"
    [age]="25"
    [city]="Beijing"
)

# 遍历键
for key in "${!user[@]}"; do
    echo "键：$key"
done

# 遍历值
for value in "${user[@]}"; do
    echo "值：$value"
done

# 遍历键值对
for key in "${!user[@]}"; do
    echo "$key = ${user[$key]}"
done
```

**字典操作**：
```bash
#!/bin/bash
declare -A user

# 添加/更新
user[name]="John"
user[name]="Jane"  # 更新

# 删除
unset user[age]

# 清空
user=()

# 合并字典
declare -A dict1=([a]=1 [b]=2)
declare -A dict2=([c]=3 [d]=4)
declare -A merged=()

for key in "${!dict1[@]}"; do
    merged[$key]=${dict1[$key]}
done
for key in "${!dict2[@]}"; do
    merged[$key]=${dict2[$key]}
done
```

**实战应用**：
```bash
#!/bin/bash
# 配置文件读取
declare -A config

while IFS='=' read -r key value; do
    [[ $key =~ ^#.*$ ]] && continue  # 跳过注释
    [[ -z $key ]] && continue         # 跳过空行
    config[$key]=$value
done < config.ini

echo "数据库：${config[db_host]}:${config[db_port]}"
```

### 字符串高级操作

**字符串拼接**：
```bash
#!/bin/bash
str1="Hello"
str2="World"

# 方法 1：直接连接
result="${str1} ${str2}"
echo $result  # Hello World

# 方法 2：追加
str1+=" World"
echo $str1    # Hello World

# 方法 3：printf
result=$(printf "%s %s" "$str1" "$str2")
```

**字符串分割**：
```bash
#!/bin/bash
str="apple,banana,cherry"

# 方法 1：IFS 分割
IFS=',' read -ra arr <<< "$str"
echo ${arr[@]}  # apple banana cherry

# 方法 2：tr 替换
echo "$str" | tr ',' '\n'  # 每行一个

# 方法 3：sed 分割
echo "$str" | sed 's/,/\n/g'
```

**字符串替换**：
```bash
#!/bin/bash
str="Hello World"

# 替换第一个
echo ${str/World/Shell}     # Hello Shell

# 替换所有
str="aaa bbb aaa"
echo ${str//aaa/xxx}        # xxx bbb xxx

# 开头替换
echo ${str/#aaa/xxx}        # xxx bbb aaa

# 结尾替换
echo ${str/%aaa/xxx}        # aaa bbb xxx

# 删除
str="  Hello  "
echo "${str// /}"           # Hello（删除所有空格）
```

**字符串清理**：
```bash
#!/bin/bash
str="  Hello World  "

# 去除两端空格（Bash 4.0+）
str="${str#"${str%%[![:space:]]*}"}"  # 左
str="${str%"${str##*[![:space:]]}"}"  # 右

# 或用 sed
str=$(echo "$str" | sed 's/^[[:space:]]*//;s/[[:space:]]*$//')

# 删除特定字符
str="abc123def"
echo ${str//[0-9]/}  # abcdef（删除数字）
```

**字符串反转**：
```bash
#!/bin/bash
str="Hello"

# 方法 1：rev 命令
echo "$str" | rev  # olleH

# 方法 2：循环
reverse=""
for ((i=${#str}-1; i>=0; i--)); do
    reverse+="${str:$i:1}"
done
echo $reverse  # olleH
```

**大小写转换**：
```bash
#!/bin/bash
str="Hello World"

# Bash 4.0+
echo ${str,,}   # hello world（小写）
echo ${str^^}   # HELLO WORLD（大写）
echo ${str^}    # Hello world（首字母大写）

# tr 命令（兼容老版本）
echo "$str" | tr '[:upper:]' '[:lower:]'
echo "$str" | tr '[:lower:]' '[:upper:]'
```

## ⚠️ 常见陷阱 ⭐ 新增

### 1. 数组不引用导致拆分

**错误**：
```bash
arr=("file 1.txt" "file 2.txt")
for f in ${arr[@]}; do  # ❌ 拆分成 4 个词
    echo "$f"
done
# 输出：file 1.txt file 2.txt
```

**正确**：
```bash
arr=("file 1.txt" "file 2.txt")
for f in "${arr[@]}"; do  # ✅ 保留空格
    echo "$f"
done
# 输出：file 1.txt
#       file 2.txt
```

---

### 2. 关联数组未声明

**错误**：
```bash
user[name]="John"  # ❌ Bash 3 会当成普通数组
```

**正确**：
```bash
declare -A user    # ✅ 先声明
user[name]="John"
```

---

### 3. 字符串分割丢失空元素

**错误**：
```bash
str="a,,b"
IFS=',' read -ra arr <<< "$str"  # ❌ 空元素丢失
```

**正确**：
```bash
str="a,,b"
IFS=',' read -ra arr <<< "$str"
# 需要手动处理空元素
```

---

### 4. 字符串替换未引用

**错误**：
```bash
str="Hello World"
echo ${str/World/Shell}  # ✅ 可以
echo ${str/$var/repl}    # ❌ var 含特殊字符会出错
```

**正确**：
```bash
str="Hello World"
var="World"
echo ${str/$var/repl}    # ✅ 变量不含特殊字符时可用
```

---

### 5. 数组索引越界

**错误**：
```bash
arr=(1 2 3)
echo ${arr[5]}  # ❌ 输出空，但不报错
```

**正确**：
```bash
arr=(1 2 3)
if [ $idx -lt ${#arr[@]} ]; then
    echo ${arr[$idx]}
else
    echo "索引越界"
fi
```

## ❓ 常见问题 FAQ ⭐ 新增

### Q1: 如何检查数组是否为空？

**方法**：
```bash
arr=()
if [ ${#arr[@]} -eq 0 ]; then
    echo "数组为空"
fi
```

---

### Q2: 关联数组如何按值排序？

**方法**：
```bash
declare -A scores=([Alice]=85 [Bob]=92 [Charlie]=78)

# 按值排序
for key in $(for k in "${!scores[@]}"; do
    echo "$k ${scores[$k]}"
done | sort -k2 -rn | awk '{print $1}'); do
    echo "$key: ${scores[$key]}"
done
```

---

### Q3: 如何从数组删除元素？

**方法**：
```bash
arr=(1 2 3 4 5)

# 删除指定索引
unset arr[2]

# 重新索引（可选）
arr=("${arr[@]}")

echo ${arr[@]}  # 1 2 4 5
```

---

### Q4: 字符串分割如何保留空元素？

**方法**：
```bash
str="a,,b,,c"

# 用 while read
echo "$str" | tr ',' '\n' | while IFS= read -r line; do
    echo "[$line]"
done
# 输出：[a] [] [b] [] [c]
```

---

### Q5: 如何检查键是否在字典中？

**方法**：
```bash
declare -A user=([name]="John")

# 方法 1：-v 测试（Bash 4.2+）
if [[ -v user[name] ]]; then
    echo "键存在"
fi

# 方法 2：检查值是否为空
if [ -n "${user[name]}" ]; then
    echo "键存在"
fi

# 方法 3：遍历键
for key in "${!user[@]}"; do
    if [ "$key" = "name" ]; then
        echo "键存在"
    fi
done
```
